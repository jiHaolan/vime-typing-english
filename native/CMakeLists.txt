cmake_minimum_required(VERSION 3.22)
project(vime_rime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)

if(NOT VIME_ENGINE_BUILD OR NOT EXISTS "${VIME_ENGINE_BUILD}/receipt.json")
    message(FATAL_ERROR "Build tools/build_native_engine.py first; provider librime.a is not accepted")
endif()
file(READ "${VIME_ENGINE_BUILD}/receipt.json" ENGINE_RECEIPT)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sources.lock.json" SOURCE_LOCK)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/sources.lock.json" SOURCE_LOCK_SHA)
string(JSON ENGINE_PASSED GET "${ENGINE_RECEIPT}" passed)
string(JSON ENGINE_LOCK_SHA GET "${ENGINE_RECEIPT}" source_lock_sha256)
string(JSON ENGINE_OCTAGRAM GET "${ENGINE_RECEIPT}" octagram_included)
string(JSON ENGINE_PLUGIN_COUNT LENGTH "${ENGINE_RECEIPT}" plugins)
string(JSON ENGINE_PLUGIN_FIRST GET "${ENGINE_RECEIPT}" plugins 0)
string(JSON ENGINE_PLUGIN_SECOND GET "${ENGINE_RECEIPT}" plugins 1)
if(NOT ENGINE_PASSED OR ENGINE_OCTAGRAM OR NOT ENGINE_LOCK_SHA STREQUAL SOURCE_LOCK_SHA
   OR NOT ENGINE_PLUGIN_COUNT EQUAL 2 OR NOT ENGINE_PLUGIN_FIRST STREQUAL "lua" OR NOT ENGINE_PLUGIN_SECOND STREQUAL "predict")
    message(FATAL_ERROR "Engine receipt does not match explicit source/plugin policy")
endif()
string(JSON ENGINE_ARCHIVE GET "${ENGINE_RECEIPT}" abis "${ANDROID_ABI}" archive)
string(JSON ENGINE_EXPECTED GET "${ENGINE_RECEIPT}" abis "${ANDROID_ABI}" sha256)
file(SHA256 "${ENGINE_ARCHIVE}" ENGINE_ACTUAL)
if(NOT ENGINE_EXPECTED STREQUAL ENGINE_ACTUAL)
    message(FATAL_ERROR "Source-built engine archive checksum differs")
endif()
string(JSON ENGINE_COMMIT GET "${ENGINE_RECEIPT}" rime_commit)
set(ENGINE_INCLUDE "${VIME_ENGINE_BUILD}/core-source/librime-${ENGINE_COMMIT}/src")
if(NOT EXISTS "${ENGINE_INCLUDE}/rime_api.h")
    message(FATAL_ERROR "Source-built public Rime API header missing")
endif()
add_library(vime_rime SHARED rime_bridge.cpp)
target_include_directories(vime_rime PRIVATE include "${ENGINE_INCLUDE}")
set(PRE "${RIME_PREBUILT}")
foreach(PACKAGE glog leveldb lua marisa opencc yaml-cpp)
    if(PACKAGE STREQUAL "lua")
        set(ARCHIVE_NAME "liblua_static.a")
    else()
        set(ARCHIVE_NAME "lib${PACKAGE}.a")
    endif()
    set(DEPENDENCY "${PACKAGE}/${ANDROID_ABI}/lib/${ARCHIVE_NAME}")
    string(JSON EXPECTED GET "${SOURCE_LOCK}" dependency_archives "${DEPENDENCY}" sha256)
    file(SHA256 "${PRE}/${DEPENDENCY}" ACTUAL)
    if(NOT EXPECTED STREQUAL ACTUAL)
        message(FATAL_ERROR "Dependency differs from pinned provider: ${DEPENDENCY}")
    endif()
endforeach()
if(NOT VIME_DISTRIBUTION_VERSION)
    set(VIME_DISTRIBUTION_VERSION "development")
endif()
target_compile_definitions(vime_rime PRIVATE VIME_DISTRIBUTION_VERSION="${VIME_DISTRIBUTION_VERSION}")
target_link_libraries(vime_rime PRIVATE
    -Wl,--whole-archive "${ENGINE_ARCHIVE}" -Wl,--no-whole-archive
    -Wl,--start-group
    "${PRE}/glog/${ANDROID_ABI}/lib/libglog.a"
    "${PRE}/leveldb/${ANDROID_ABI}/lib/libleveldb.a"
    "${PRE}/lua/${ANDROID_ABI}/lib/liblua_static.a"
    "${PRE}/marisa/${ANDROID_ABI}/lib/libmarisa.a"
    "${PRE}/opencc/${ANDROID_ABI}/lib/libopencc.a"
    "${PRE}/yaml-cpp/${ANDROID_ABI}/lib/libyaml-cpp.a"
    -Wl,--end-group log android)
target_link_options(vime_rime PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--build-id=sha1")
