cmake_minimum_required(VERSION 3.22)
project(vime_xasr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
if(NOT CRISPASR_SOURCE)
 message(FATAL_ERROR "CRISPASR_SOURCE is required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_compile_definitions(GGML_MAX_NAME=128)
add_subdirectory(${CRISPASR_SOURCE}/ggml ggml)
add_library(vime_xasr SHARED jni.cpp xasr_adapter.cpp
 ${CRISPASR_SOURCE}/src/core/gguf_loader.cpp
 ${CRISPASR_SOURCE}/src/core/kaldi_fbank.cpp)
target_include_directories(vime_xasr PRIVATE ${CRISPASR_SOURCE}/src ${CRISPASR_SOURCE}/ggml/src)
target_link_libraries(vime_xasr PRIVATE ggml ggml-base ggml-cpu)
# C++ objects stay within this JNI library. Do not replace the existing Rime runtime.
target_link_options(vime_xasr PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
