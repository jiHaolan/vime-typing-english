cmake_minimum_required(VERSION 3.22)
project(jianyi_xasr_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CRISPASR_SOURCE)
 message(FATAL_ERROR "Set CRISPASR_SOURCE to the pinned source checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_compile_definitions(GGML_MAX_NAME=128)
add_subdirectory(${CRISPASR_SOURCE}/ggml ggml)
add_library(jianyi_xasr_core STATIC
 ${CRISPASR_SOURCE}/src/xasr.cpp
 ${CRISPASR_SOURCE}/src/core/gguf_loader.cpp
 ${CRISPASR_SOURCE}/src/core/kaldi_fbank.cpp)
target_include_directories(jianyi_xasr_core PUBLIC ${CRISPASR_SOURCE}/src PRIVATE ${CRISPASR_SOURCE}/ggml/src)
target_link_libraries(jianyi_xasr_core PUBLIC ggml ggml-base ggml-cpu)
add_executable(jianyi-xasr-probe probe.cpp)
target_link_libraries(jianyi-xasr-probe PRIVATE jianyi_xasr_core)
target_link_options(jianyi-xasr-probe PRIVATE -Wl,-z,max-page-size=16384)
