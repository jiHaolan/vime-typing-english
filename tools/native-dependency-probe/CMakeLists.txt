cmake_minimum_required(VERSION 3.22)
project(vime_dependency_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT VIME_DEPENDENCY_BUILD OR NOT EXISTS "${VIME_DEPENDENCY_BUILD}/receipt.json")
  message(FATAL_ERROR "Verified dependency source build required")
endif()
file(READ "${VIME_DEPENDENCY_BUILD}/receipt.json" RECEIPT)
string(JSON PASSED GET "${RECEIPT}" passed)
string(JSON PROVIDER_USED GET "${RECEIPT}" provider_binary_inputs_used)
if(NOT PASSED OR PROVIDER_USED)
  message(FATAL_ERROR "Source-only dependency receipt required")
endif()
add_executable(vime_dependency_probe main.cpp)
target_compile_definitions(vime_dependency_probe PRIVATE GLOG_USE_GLOG_EXPORT GLOG_STATIC_DEFINE YAML_CPP_STATIC_DEFINE)
foreach(PACKAGE glog leveldb lua marisa opencc yaml-cpp)
  set(PREFIX "${VIME_DEPENDENCY_BUILD}/prefix/${PACKAGE}/${ANDROID_ABI}")
  string(JSON ARCHIVE GET "${RECEIPT}" builds "${PACKAGE}/${ANDROID_ABI}" archive)
  string(JSON EXPECTED GET "${RECEIPT}" builds "${PACKAGE}/${ANDROID_ABI}" sha256)
  set(LIBRARY "${VIME_DEPENDENCY_BUILD}/prefix/${ARCHIVE}")
  file(SHA256 "${LIBRARY}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Dependency source-build digest differs: ${PACKAGE}")
  endif()
  target_include_directories(vime_dependency_probe PRIVATE "${PREFIX}/include")
  target_link_libraries(vime_dependency_probe PRIVATE "${LIBRARY}")
endforeach()
target_link_libraries(vime_dependency_probe PRIVATE log android dl m)
target_link_options(vime_dependency_probe PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/link.map")
