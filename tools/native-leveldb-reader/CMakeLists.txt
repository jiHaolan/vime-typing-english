cmake_minimum_required(VERSION 3.22)
project(vime_leveldb_reader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../native/sources.lock.json" SOURCE_LOCK)
if(VIME_DEPENDENCY_BUILD)
    file(READ "${VIME_DEPENDENCY_BUILD}/receipt.json" DEPENDENCY_RECEIPT)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../native/sources.lock.json" LOCK_SHA)
    string(JSON DEPENDENCY_PASSED GET "${DEPENDENCY_RECEIPT}" passed)
    string(JSON DEPENDENCY_LOCK GET "${DEPENDENCY_RECEIPT}" source_lock_sha256)
    string(JSON PROVIDER_USED GET "${DEPENDENCY_RECEIPT}" provider_binary_inputs_used)
    if(NOT DEPENDENCY_PASSED OR PROVIDER_USED OR NOT DEPENDENCY_LOCK STREQUAL LOCK_SHA)
        message(FATAL_ERROR "Matching verified dependency source build required")
    endif()
    string(JSON EXPECTED GET "${DEPENDENCY_RECEIPT}" builds "leveldb/${ANDROID_ABI}" sha256)
else()
    # Historical0.23.1 receipts use the original authenticated provider archive.
    string(JSON EXPECTED GET "${SOURCE_LOCK}" dependency_archives "leveldb/${ANDROID_ABI}/lib/libleveldb.a" sha256)
endif()
file(SHA256 "${VIME_LEVELDB_ROOT}/lib/libleveldb.a" ACTUAL)
if(NOT EXPECTED STREQUAL ACTUAL)
    message(FATAL_ERROR "Pinned LevelDB archive differs from the selected build receipt")
endif()
add_executable(vime_leveldb_reader main.cpp)
target_include_directories(vime_leveldb_reader PRIVATE "${VIME_LEVELDB_ROOT}/include")
target_link_libraries(vime_leveldb_reader PRIVATE "${VIME_LEVELDB_ROOT}/lib/libleveldb.a" log)
target_link_options(vime_leveldb_reader PRIVATE "-Wl,-z,max-page-size=16384")
